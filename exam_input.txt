int main() {
    int a, m, total, i, j;

    a = 10;
    m = 25;
    total = a + m * 2;

    if (a < m) {
        total = total + 5;
    } else {
        total = total - 5;
    }

    for (i = 0; i < 5; i++) {
        total = total + i;

        if (i < 3) {
            total = total + 2;
        } else {
            total = total - 1;
        }
    }

    j = 0;

    while (j < 3) {
        {
            total = total + j;
            j = j + 1;
        }
    }

    return total;
}
